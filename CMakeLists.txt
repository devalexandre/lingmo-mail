cmake_minimum_required(VERSION 3.16)
project(lingmo-mail VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Qml Quick DBus)
add_executable(lingmo-mail src/main.cpp src/mailbridge.cpp resources.qrc
    src/calendarbridge.cpp src/reminderservice.cpp)
target_link_libraries(lingmo-mail PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml Qt6::Quick Qt6::DBus)
target_compile_definitions(lingmo-mail PRIVATE MAIL_BACKEND="${CMAKE_INSTALL_FULL_LIBEXECDIR}/lingmo-mail/backend.py")
install(TARGETS lingmo-mail RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS backend/backend.py DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/lingmo-mail)
target_compile_definitions(lingmo-mail PRIVATE CALENDAR_BACKEND="${CMAKE_INSTALL_FULL_LIBEXECDIR}/lingmo-mail/calendar_backend.py")
install(PROGRAMS backend/calendar_backend.py DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/lingmo-mail)
install(FILES backend/ics.py DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/lingmo-mail)
install(FILES data/lingmo-mail-reminders.desktop DESTINATION ${CMAKE_INSTALL_FULL_SYSCONFDIR}/xdg/autostart)
install(FILES data/lingmo-mail.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/lingmo-mail.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME mail-backend COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests)
